library(tidyverse)
library(stringr)

# Cargamos los datos
setwd("C:/DataEdelgado/Encar/DocEducacion/Curso2025-26/PIA/Tema5 - Introducción a R/UT5_EJ5_MusicaR")
datos <- read_csv("artists.csv")

#0. Limpieza de datos
# Eliminamos filas con datos faltantes en las columnas principales
datos_sin_na <- filter(datos, !is.na(artist_mb) & !is.na(artist_lastfm) & !is.na(country_mb) & !is.na(country_lastfm))

# Eliminamos filas con datos en idiomas o caracteres no identificables
# Usamos una expresión regular para filtrar solo caracteres permitidos
expresion_regular <- "^[\\p{L}0-9 .,;:'\"!?&()\\-]+$" # Esta expresión permite letras de cualquier idioma, números y algunos símbolos comunes.
datos_limpios <- filter(datos_sin_na, str_detect(artist_mb, expresion_regular) &
                          str_detect(artist_lastfm, expresion_regular) &
                          str_detect(country_mb, expresion_regular) &
                          str_detect(country_lastfm, expresion_regular))

# Guardar los datos limpios
write_csv(datos_limpios, "C:/DataEdelgado/Encar/DocEducacion/AI&BD/PIA/Contenidos2023-2024/Tema5 - Machine Learning y R/UT5_Ejercicio3_MusicaR/datos_limpios.csv")


# Contar el número total de artistas únicos
total_artistas <- n_distinct(datos_limpios$artist_lastfm)

# Calcular estadísticas descriptivas para oyentes y reproducciones
media_oyentes <- mean(datos_limpios$listeners_lastfm, na.rm = TRUE)
mediana_oyentes <- median(datos_limpios$listeners_lastfm, na.rm = TRUE)
desv_std_oyentes <- sd(datos_limpios$listeners_lastfm, na.rm = TRUE)

media_scrobbles <- mean(datos_limpios$scrobbles_lastfm, na.rm = TRUE)
mediana_scrobbles <- median(datos_limpios$scrobbles_lastfm, na.rm = TRUE)
desv_std_scrobbles <- sd(datos_limpios$scrobbles_lastfm, na.rm = TRUE)

# Identificar los países con más artistas representados
paises_conteo <- count(datos_limpios, country_lastfm, sort = TRUE)

# Seleccionar los top 10 países con más artistas representados para el gráfico
top_paises <- head(paises_conteo, 10)
top_paises


# Representar gráficamente los países con más artistas representados
ggplot(data = top_paises, aes(x = reorder(country_lastfm, n), y = n)) +
  geom_bar(stat = "identity") +
  theme_minimal() +
  labs(title = "Top 10 Países con Más Artistas Representados",
       x = "País",
       y = "Número de Artistas") +
  coord_flip()

#Contar artistas por país
conteo_artistas_pais <- count(datos_limpios, country_lastfm, sort = TRUE)

#Calcular el total de artistas
total_artistas <- sum(conteo_artistas_pais$n)

#Añadir una columna de porcentaje al dataframe
conteo_artistas_pais$porcentaje <- (conteo_artistas_pais$n / total_artistas) * 100

# Filtrar países con al menos 5% de representación
paises_filtrados <- filter(conteo_artistas_pais, porcentaje >= 3)


ggplot(data = paises_filtrados, aes(x = reorder(country_lastfm, n), y = n, fill = country_lastfm)) +
  geom_bar(stat = "identity") +
  scale_fill_manual(values = RColorBrewer::brewer.pal(n = 9, name = "Pastel1")) +
  labs(title = "Distribución de Artistas por País", x = "País", y = "Número de Artistas") +
  theme(axis.text.x = element_text(angle = 90, hjust = 1))

ggplot(data = paises_filtrados, aes(x = "", y = n, fill = country_lastfm)) +
  geom_bar(width = 1, stat = "identity") +
  scale_fill_manual(values = RColorBrewer::brewer.pal(n = 9, name = "Pastel1")) +
  coord_polar("y") +
  labs(title = "Distribución de Artistas por País", x = "", y = "", fill = "País") +
  theme_void()
